cmake_minimum_required(VERSION 3.20)
project(uhntiss_pocket VERSION 0.6.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(POCKET_SANITIZE "Address and undefined behaviour sanitizers" OFF)
add_library(pocket STATIC src/engine.cpp)
target_include_directories(pocket PUBLIC include)
set_target_properties(pocket PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_options(pocket PRIVATE -Wall -Wextra -Wpedantic)
if(POCKET_SANITIZE)
  target_compile_options(pocket PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(pocket PUBLIC -fsanitize=address,undefined)
endif()
add_executable(pocket_tests tests/engine_tests.cpp)
find_package(Threads REQUIRED)
target_link_libraries(pocket_tests PRIVATE pocket Threads::Threads)
add_executable(pocket_render tools/render.cpp)
target_link_libraries(pocket_render PRIVATE pocket)
add_executable(pocket_analyze tools/analyze.cpp)
target_link_libraries(pocket_analyze PRIVATE pocket)
add_executable(pocket_stream tools/stream.cpp)
target_link_libraries(pocket_stream PRIVATE pocket)
add_executable(pocket_bench tools/bench.cpp)
target_link_libraries(pocket_bench PRIVATE pocket)
add_executable(pocket_follow tools/follow.cpp)
target_link_libraries(pocket_follow PRIVATE pocket)
add_executable(pocket_classify tools/classify.cpp)
target_link_libraries(pocket_classify PRIVATE pocket)
add_executable(pocket_profile tools/profile.cpp)
target_link_libraries(pocket_profile PRIVATE pocket)
add_executable(pocket_enrollment_tests tests/enrollment_tests.cpp)
target_link_libraries(pocket_enrollment_tests PRIVATE pocket)
enable_testing()
add_test(NAME pocket_engine COMMAND pocket_tests)
add_test(NAME pocket_enrollment COMMAND pocket_enrollment_tests)
set(MAX_SDK_PATH "" CACHE PATH "Cycling74/max-sdk-base checkout")
if(MAX_SDK_PATH AND APPLE)
  add_library(uhntiss_pocket MODULE max/external.cpp)
  target_include_directories(uhntiss_pocket PRIVATE "${MAX_SDK_PATH}/c74support/max-includes" "${MAX_SDK_PATH}/c74support/msp-includes")
  target_link_libraries(uhntiss_pocket PRIVATE pocket "${MAX_SDK_PATH}/c74support/msp-includes/MaxAudioAPI.framework")
  target_link_options(uhntiss_pocket PRIVATE "-Wl,-undefined,dynamic_lookup")
  set_target_properties(uhntiss_pocket PROPERTIES PREFIX "" OUTPUT_NAME "uhntiss.pocket06~" BUNDLE TRUE BUNDLE_EXTENSION "mxo" MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/max/Info.plist.in")
  add_custom_command(TARGET uhntiss_pocket POST_BUILD COMMAND /usr/bin/codesign --force --sign - "$<TARGET_BUNDLE_DIR:uhntiss_pocket>" VERBATIM)
endif()
